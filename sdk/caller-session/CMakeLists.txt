cmake_minimum_required(VERSION 3.16)
project(fx_caller_session LANGUAGES C)
add_library(fx_caller_session STATIC src/fx_caller_session.c)
target_include_directories(fx_caller_session PUBLIC include)
target_compile_features(fx_caller_session PUBLIC c_std_99)
add_library(FxStudio::CallerSession ALIAS fx_caller_session)
option(FXCS_BUILD_HOST_TESTS "Build host lifecycle contract tests" OFF)
if(FXCS_BUILD_HOST_TESTS)
    enable_testing()
    add_executable(fx_caller_session_test tests/session_test.c)
    target_link_libraries(fx_caller_session_test PRIVATE FxStudio::CallerSession)
    add_test(NAME fx_caller_session_contract COMMAND fx_caller_session_test)
endif()
