cmake_minimum_required(VERSION 3.15)
project(KhiMap C ASM)
add_subdirectory(vendor/fxlarge)
include(GenerateG1A)
if(NOT Python3_EXECUTABLE)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
endif()
add_executable(khimap src/start.S src/main.c src/sha256.c src/resident.c src/resident_syscalls.S src/vectors.S src/gateway.S)
target_link_libraries(khimap PRIVATE FxStudio::LargeRuntime)
target_compile_options(khimap PRIVATE -m4-nofpu -Os -ffreestanding -fno-builtin -fno-stack-protector -fno-unwind-tables -fno-asynchronous-unwind-tables -ffunction-sections -fdata-sections -Wall -Wextra -Werror)
target_link_options(khimap PRIVATE -m4-nofpu -T "${CMAKE_CURRENT_SOURCE_DIR}/src/loader.ld" -Wl,--emit-relocs,-Map=khimap.map)
set_property(TARGET khimap APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/loader.ld")
add_custom_command(TARGET khimap POST_BUILD COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/relocations.py" fixups "$<TARGET_FILE:khimap>")
generate_g1a(TARGET khimap OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/ExtLoad.g1a" NAME "ExtLoad" INTERNAL "@EXTLOAD" ICON "assets/icon.png")
