cmake_minimum_required(VERSION 3.15)
project(MappedLoader C ASM)
add_subdirectory(vendor/fxlarge)
include(GenerateG1A)
if(NOT Python3_EXECUTABLE)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
endif()
add_executable(kload src/start.S src/main.c src/resident.c src/resident_syscalls.S src/callback.S src/gateway.S src/sha256.c)
target_link_libraries(kload PRIVATE FxStudio::LargeRuntime)
target_compile_options(kload PRIVATE -m4-nofpu -Os -ffreestanding -fno-builtin -fno-stack-protector -fno-unwind-tables -fno-asynchronous-unwind-tables -ffunction-sections -fdata-sections -Wall -Wextra)
target_link_options(kload PRIVATE -m4-nofpu -T "${CMAKE_CURRENT_SOURCE_DIR}/src/probe.ld" -Wl,--emit-relocs,-Map=kload.map)
set_property(TARGET kload APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/probe.ld")
add_custom_command(TARGET kload POST_BUILD COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/relocations.py" fixups "$<TARGET_FILE:kload>")
generate_g1a(TARGET kload OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/KLoad.g1a" NAME "KLoad" INTERNAL "@KLOAD" ICON "assets/icon.png")
