cmake_minimum_required(VERSION 3.15)
project(LargeGuest C ASM)
include(GenerateG1A)
add_subdirectory(vendor/fxlarge)
add_executable(largeguest start.S main.c)
target_link_libraries(largeguest PRIVATE FxStudio::LargeRuntime)
target_compile_options(largeguest PRIVATE -m4-nofpu -Os -ffreestanding -fno-builtin -Wall -Wextra -Werror)
target_link_options(largeguest PRIVATE -m4-nofpu -T "${CMAKE_CURRENT_SOURCE_DIR}/guest.ld")
set_property(TARGET largeguest APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/guest.ld")
generate_g1a(TARGET largeguest OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/LargeGuest.g1a" NAME "Guest" INTERNAL "@FXGUEST" ICON "assets/icon.png")
