cmake_minimum_required(VERSION 3.15)
project(Map698 C ASM)
include(GenerateG1A)
if(NOT Python3_EXECUTABLE)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
endif()
add_executable(map698 src/start.S src/main.c src/resident.c src/resident_syscalls.S src/vectors.S)
target_compile_options(map698 PRIVATE -m4-nofpu -Os -ffreestanding -fno-builtin -fno-stack-protector -fno-unwind-tables -fno-asynchronous-unwind-tables -ffunction-sections -fdata-sections -Wall -Wextra)
target_link_options(map698 PRIVATE -m4-nofpu -T "${CMAKE_CURRENT_SOURCE_DIR}/src/probe.ld" -Wl,--emit-relocs,-Map=map698.map)
set_property(TARGET map698 APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/probe.ld")
add_custom_command(TARGET map698 POST_BUILD COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/relocations.py" fixups "$<TARGET_FILE:map698>")
generate_g1a(TARGET map698 OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/Map698.g1a" NAME "Map698" INTERNAL "@MAP698" ICON "assets/icon.png")
add_custom_command(TARGET map698 POST_BUILD COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/bundle.py" "${CMAKE_CURRENT_BINARY_DIR}")
