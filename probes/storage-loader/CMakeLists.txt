cmake_minimum_required(VERSION 3.15)
project(FlashLoader C ASM)
include(GenerateG1A)
if(NOT Python3_EXECUTABLE)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
endif()
add_executable(flashloader src/start.S src/main.c src/resident.c src/resident_syscalls.S src/callback.S src/gateway.S src/sha256.c)
target_compile_options(flashloader PRIVATE -m4-nofpu -Os -ffreestanding -fno-builtin -fno-stack-protector -fno-unwind-tables -fno-asynchronous-unwind-tables -ffunction-sections -fdata-sections -Wall -Wextra -Werror)
target_link_options(flashloader PRIVATE -m4-nofpu -T "${CMAKE_CURRENT_SOURCE_DIR}/src/probe.ld" -Wl,--emit-relocs,-Map=flashloader.map)
set_property(TARGET flashloader APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/probe.ld")
add_custom_command(TARGET flashloader POST_BUILD COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/relocations.py" fixups "$<TARGET_FILE:flashloader>")
generate_g1a(TARGET flashloader OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/FlashLd.g1a" NAME "FlashLd" INTERNAL "@FLASHLD" ICON "assets/icon.png")
