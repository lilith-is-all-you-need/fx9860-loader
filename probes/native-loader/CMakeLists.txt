cmake_minimum_required(VERSION 3.15)
project(HotLoader C ASM)
include(GenerateG1A)
if(NOT Python3_EXECUTABLE)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
endif()
add_executable(loader src/start.S src/main.c src/resident.c src/resident_syscalls.S src/callback.S src/gateway.S src/sha256.c)
target_compile_options(loader PRIVATE -m4-nofpu -Os -ffreestanding -fno-builtin -fno-stack-protector -fno-unwind-tables -fno-asynchronous-unwind-tables -ffunction-sections -fdata-sections -Wall -Wextra)
target_link_options(loader PRIVATE -m4-nofpu -T "${CMAKE_CURRENT_SOURCE_DIR}/src/probe.ld" -Wl,--emit-relocs,-Map=loader.map)
set_property(TARGET loader APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/probe.ld")
add_custom_command(TARGET loader POST_BUILD COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/pack.py" fixups "$<TARGET_FILE:loader>")
generate_g1a(TARGET loader OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/Loader.g1a" NAME "Loader" INTERNAL "@HOTLOAD" ICON "assets/icon.png")
